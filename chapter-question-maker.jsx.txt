const CLASSES = ["6", "7", "8", "9", "10", "11", "12"];
const LANGS = ["English", "Assamese (অসমীয়া)", "Hindi (हिन्दी)"];

const TYPES = [
  { id: "mcq", label: "MCQ", count: 6 },
  { id: "ar", label: "Assertion & Reason", count: 4 },
  { id: "concept", label: "Concept based", count: 5 },
  { id: "short", label: "Short answer", count: 5 },
  { id: "long", label: "Long answer", count: 3 },
  { id: "case", label: "Case based", count: 2 },
];

const AR_OPTIONS = [
  "A. Both A and R are true, and R is the correct explanation of A.",
  "B. Both A and R are true, but R is not the correct explanation of A.",
  "C. A is true, but R is false.",
  "D. A is false, but R is true.",
];

const TAGS = {
  knowledge: "Knowledge",
  understanding: "Understanding",
  application: "Application",
  skill: "Skill",
};

const TYPE_RULES = {
  mcq: `Each item: {"q": string, "options": [4 strings without letter prefixes], "answer": "A"|"B"|"C"|"D", "explain": string (1-2 sentences), "tag": "knowledge"|"understanding"|"application"|"skill", "why": string (short)}`,
  ar: `Each item: {"assertion": string, "reason": string, "answer": "A"|"B"|"C"|"D", "explain": string (1-2 sentences), "tag": "knowledge"|"understanding"|"application"|"skill", "why": string (short)}. The answer letter follows the standard options: A both true and R explains A; B both true but R does not explain A; C A true R false; D A false R true.`,
  concept: `Each item: {"q": string (tests understanding, application or reasoning, not memory), "answer": string (concise model answer), "tag": "knowledge"|"understanding"|"application"|"skill", "why": string (short)}`,
  short: `Each item: {"q": string (2-3 mark question), "answer": string (model answer with key points), "tag": "knowledge"|"understanding"|"application"|"skill", "why": string (short)}`,
  long: `Each item: {"q": string (5 mark question), "answer": string (model answer with key points), "tag": "knowledge"|"understanding"|"application"|"skill", "why": string (short)}`,
  case: `Each item: {"q": string (a short passage or situation followed by 2 sub-questions), "answer": string (answers to the sub-questions), "tag": "knowledge"|"understanding"|"application"|"skill", "why": string (short)}`,
};

function fileToBase64(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onerror = () => reject(new Error("Could not read photo"));
    reader.onload = () => {
      const img = new Image();
      img.onerror = () => reject(new Error("Could not open photo"));
      img.onload = () => {
        const max = 1400;
        const scale = Math.min(1, max / Math.max(img.width, img.height));
        const canvas = document.createElement("canvas");
        canvas.width = Math.round(img.width * scale);
        canvas.height = Math.round(img.height * scale);
        canvas.getContext("2d").drawImage(img, 0, 0, canvas.width, canvas.height);
        const url = canvas.toDataURL("image/jpeg", 0.8);
        resolve({ data: url.split(",")[1], preview: url });
      };
      img.src = reader.result;
    };
    reader.readAsDataURL(file);
  });
}

async function generate({ type, count, images, text, board, cls, subject, lang }) {
  const system = `You are an experienced ${board} exam paper setter and teacher. You read textbook chapter photos and write exam questions.
Return ONLY a JSON array with exactly ${count} items. No markdown, no commentary.
${TYPE_RULES[type]}
Rules:
- Base every question strictly on the chapter content visible in the photos.
- Match the style and difficulty of ${board} Class ${cls}${subject ? " " + subject : ""} board papers.
- ${board.startsWith("SEBA") ? (cls === "10" ? "Follow the CURRENT SEBA HSLC question paper pattern (from 2025 onwards): 90 marks theory, with OMR-based MCQs plus short answer and long answer questions, set across Knowledge, Understanding, Application and Skill categories. Use only the present pattern, not old papers." : cls === "9" ? "Follow the present SEBA Class 9 annual examination style: MCQs, short answer and long answer questions across Knowledge, Understanding, Application and Skill categories, with Internal Assessment separate. Do not invent exact marks or question counts. Use only the present style, not old papers." : "Follow the present SEBA question style for this class, not old papers.") : "Follow the current official paper pattern of " + board + ", not old papers."}
- "tag": the competency category this question tests in the current blueprint (knowledge = recall of facts and definitions; understanding = explain or interpret; application = use in a new situation; skill = diagrams, calculation, graphs, maps or practical skills). Use a sensible mix.
- "why": one short line on why this question fits the current paper pattern (the question style or the important concept it covers).
- Write all text in ${lang}.`;

  const content = [
    ...images.map((im) => ({
      type: "image",
      source: { type: "base64", media_type: "image/jpeg", data: im.data },
    })),
    {
      type: "text",
      text:
        (text && text.trim() ? `Chapter text:\n${text.trim().slice(0, 6000)}\n\n` : "") +
        `Create ${count} important ${type} questions from this chapter.`,
    },
  ];

  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-6",
      max_tokens: 1000,
      system,
      messages: [{ role: "user", content }],
    }),
  });
  const data = await response.json();
  const out = (data.content || []).map((c) => c.text || "").join("");
  const clean = out.replace(/```json|```/g, "").trim();
  const start = clean.indexOf("[");
  const end = clean.lastIndexOf("]");
  return JSON.parse(clean.slice(start, end + 1));
}

export default function App() {
  const [images, setImages] = useState([]);
  const [board, setBoard] = useState(BOARDS[0]);
  const [cls, setCls] = useState("9");
  const [subject, setSubject] = useState("");
  const [lang, setLang] = useState(LANGS[0]);
  const [pasted, setPasted] = useState("");
  const [selected, setSelected] = useState(["mcq", "ar", "concept"]);
  const [results, setResults] = useState({});
  const [status, setStatus] = useState({});
  const [active, setActive] = useState(null);
  const [open, setOpen] = useState({});
  const [error, setError] = useState("");

  const addPhotos = async (e) => {
    const files = Array.from(e.target.files || []).slice(0, 4 - images.length);
    try {
      const done = await Promise.all(files.map(fileToBase64));
      setImages((prev) => [...prev, ...done].slice(0, 4));
      setError("");
    } catch (err) {
      setError(err.message);
    }
    e.target.value = "";
  };

  const toggleType = (id) =>
    setSelected((s) => (s.includes(id) ? s.filter((x) => x !== id) : [...s, id]));

  const run = async (typeIds) => {
    if (!images.length && pasted.trim().length < 50) {
      setError("Chapter r photo upload koribo, nohole lekha paste koribo.");
      return;
    }
    if (!typeIds.length) {
      setError("Kimaan ek dhoroniyor question bachi lobo.");
      return;
    }
    setError("");
    setActive(typeIds[0]);
    setStatus((s) => ({ ...s, ...Object.fromEntries(typeIds.map((t) => [t, "loading"])) }));
    await Promise.all(
      typeIds.map(async (id) => {
        const t = TYPES.find((x) => x.id === id);
        try {
          const items = await generate({ type: id, count: t.count, images, text: pasted, board, cls, subject, lang });
          setResults((r) => ({ ...r, [id]: [...(r[id] || []), ...items] }));
          setStatus((s) => ({ ...s, [id]: "done" }));
        } catch (err) {
          setStatus((s) => ({ ...s, [id]: "error" }));
        }
      })
    );
  };

  const shownTypes = TYPES.filter((t) => status[t.id]);
  const current = active || (shownTypes[0] && shownTypes[0].id);
  const items = (current && results[current]) || [];

  return (
    <div className="app">
      <style>{css}</style>

      <header className="top">
        <h1>Chapter Question Maker</h1>
        <p>Chapter r photo dia. SEBA r present question paper pattern r logot important questions pabo.</p>
      </header>

      <section className="panel">
        <h2>1. Chapter r photo</h2>
        <div className="thumbs">
          {images.map((im, i) => (
            <div className="thumb" key={i}>
              <img src={im.preview} alt={`Chapter page ${i + 1}`} />
              <button
                className="x"
                aria-label="Remove photo"
                onClick={() => setImages(images.filter((_, j) => j !== i))}
              >
                ×
              </button>
            </div>
          ))}
          {images.length < 4 && (
            <>
              <label className="add">
                <span>Gallery</span>
                <input type="file" accept="image/*" multiple onChange={addPhotos} />
              </label>
              <label className="add">
                <span>Camera</span>
                <input type="file" accept="image/*" capture="environment" onChange={addPhotos} />
              </label>
            </>
          )}
        </div>
        <p className="hint">Sorrote 4 khon page. Lekha saaf dekhuwa photo dibo.</p>
        <label className="textlabel">
          Photo nuwarile chapter r lekha ইয়াত paste koribo
          <textarea
            value={pasted}
            onChange={(e) => setPasted(e.target.value)}
            rows={4}
            placeholder="Chapter r lekha ইয়াত paste koribo..."
          />
        </label>
      </section>

      <section className="panel">
        <h2>2. Board aru class</h2>
        <label>
          Board
          <select value={board} onChange={(e) => setBoard(e.target.value)}>
            {BOARDS.map((b) => (
              <option key={b}>{b}</option>
            ))}
          </select>
        </label>
        <div className="row">
          <label>
            Class
            <select value={cls} onChange={(e) => setCls(e.target.value)}>
              {CLASSES.map((c) => (
                <option key={c}>{c}</option>
              ))}
            </select>
          </label>
          <label>
            Subject
            <input
              value={subject}
              onChange={(e) => setSubject(e.target.value)}
              placeholder="Science, History..."
            />
          </label>
        </div>
        <label>
          Question r bhasha
          <select value={lang} onChange={(e) => setLang(e.target.value)}>
            {LANGS.map((l) => (
              <option key={l}>{l}</option>
            ))}
          </select>
        </label>
      </section>

      <section className="panel">
        <h2>3. Question r dhoron</h2>
        <div className="chips">
          {TYPES.map((t) => (
            <button
              key={t.id}
              className={"chip" + (selected.includes(t.id) ? " on" : "")}
              aria-pressed={selected.includes(t.id)}
              onClick={() => toggleType(t.id)}
            >
              {t.label}
            </button>
          ))}
        </div>
        <button className="go" onClick={() => run(selected)}>
          Questions bonua
        </button>
        {error && <p className="err">{error}</p>}
      </section>

      {shownTypes.length > 0 && (
        <section className="results">
          <div className="tabs" role="tablist">
            {shownTypes.map((t) => (
              <button
                key={t.id}
                role="tab"
                aria-selected={current === t.id}
                className={"tab" + (current === t.id ? " on" : "")}
                onClick={() => setActive(t.id)}
              >
                {t.label}
                {status[t.id] === "loading" && " …"}
              </button>
            ))}
          </div>

          {status[current] === "loading" && items.length === 0 && (
            <p className="state">Questions bonai ase. Alpo somoi lagibo…</p>
          )}
          {status[current] === "error" && (
            <p className="state">
              Questions bonabo nuwarilu. Photo saaf ase neki check kori abar try koribo.
            </p>
          )}

          {items.map((it, i) => {
            const key = `${current}-${i}`;
            const shown = open[key];
            return (
              <article className="card" key={key}>
                <div className="meta">
                  <span className="qno">Q{i + 1}</span>
                  {it.tag && TAGS[it.tag] && (
                    <span className={"tag " + it.tag}>{TAGS[it.tag]}</span>
                  )}
                </div>

                {current === "ar" ? (
                  <>
                    <p>
                      <b>Assertion (A):</b> {it.assertion}
                    </p>
                    <p>
                      <b>Reason (R):</b> {it.reason}
                    </p>
                    <ul className="opts">
                      {AR_OPTIONS.map((o) => (
                        <li key={o}>{o}</li>
                      ))}
                    </ul>
                  </>
                ) : (
                  <p className="q">{it.q}</p>
                )}

                {current === "mcq" && (
                  <ul className="opts">
                    {(it.options || []).map((o, k) => (
                      <li
                        key={k}
                        className={shown && "ABCD"[k] === it.answer ? "right" : ""}
                      >
                        {"ABCD"[k]}. {o}
                      </li>
                    ))}
                  </ul>
                )}

                <button
                  className="reveal"
                  onClick={() => setOpen((o) => ({ ...o, [key]: !o[key] }))}
                >
                  {shown ? "Answer lukuwa" : "Answer dekhuwa"}
                </button>

                {shown && (
                  <div className="ans">
                    <p>
                      <b>Answer:</b>{" "}
                      {current === "mcq" || current === "ar" ? it.answer : it.answer}
                    </p>
                    {it.explain && <p>{it.explain}</p>}
                    {it.why && <p className="why">{it.why}</p>}
                  </div>
                )}
                {!shown ? null : null}
              </article>
            );
          })}

          {items.length > 0 && status[current] !== "loading" && (
            <button className="more" onClick={() => run([current])}>
              Aru questions bonua
            </button>
          )}

          <p className="note">
            Tag bor SEBA r present blueprint (Knowledge, Understanding, Application, Skill)
            anusare AI-e diya. Prottek subject r exact marks aru question r sonkhya alag
            hoi pare, sutorang sebaonline.org t thaka official blueprint aru syllabus
            check koribo.
          </p>
        </section>
      )}
    </div>
  );
}
